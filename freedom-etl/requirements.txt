# freedom-etl — зависимости сервиса сбора котировок.
#
# Соглашение то же, что в корневом requirements.txt: это ЧЕЛОВЕЧЕСКИЙ
# intent-файл — нижняя граница по причине, верхняя по major-капу.
# Верхний кап не украшение: он ловит мажорный релиз, ломающий API, ДО прода.

# Асинхронный HTTP-клиент к Tradernet.
# Нижняя граница 3.14.1 не произвольна: в ветке 3.9.x живут неустранимые
# уязвимости (корневой requirements.txt документирует разбор), а 3.10.11 —
# самый ранний фикс. Держим тот же пол, что и основное приложение.
aiohttp>=3.14.1,<4.0

# Драйвер PostgreSQL. Нужен именно asyncpg, а не psycopg: executemany в нём
# идёт одним pipelined-раундом, и запись батча из 50×250 свечей укладывается
# в один round-trip вместо 12 500.
asyncpg>=0.30,<1.0

# Планировщик.
# 🔴 Кап <4.0 ОБЯЗАТЕЛЕН. APScheduler 4.x — переписанный API: вместо
# AsyncIOScheduler.add_job() там AsyncScheduler.add_schedule() с другой
# сигнатурой и другим жизненным циклом. Обновление мажора без правки main.py
# уронит контейнер на старте, а не в тестах.
APScheduler>=3.11,<4.0

# База часовых поясов IANA.
# Обязательна: slim-образы Python приезжают без системного tzdata, и
# ZoneInfo("America/New_York") падает с ZoneInfoNotFoundError. Торговые даты
# US/KZ считаются именно по ней (см. etl_processor.exchange_timezone).
tzdata>=2024.1

# Чтение .env при локальном запуске. В контейнере переменные приходят через
# env_file docker-compose, поэтому импорт в main.py необязательный.
python-dotenv>=1.0,<2.0
